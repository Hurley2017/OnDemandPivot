#!/usr/bin/env python3
r"""
OnDemandPivot - fetch the latest update from GitHub and apply it.

Save this file (any name ending in .txt or .py) and run it:

    python fetch-update.txt                       # updates .\OnDemandPivot
    python fetch-update.txt "C:\path\to\OnDemandPivot"

It tries three GitHub addresses in turn, because they are three different
hostnames and a company proxy can easily allow one and not the others. It
checks that what came back really is a bundle before running it, then hands
over to the bundle's own unpacker.

Nothing is written anywhere except the project folder you name and a copy of
the download in your temp directory.
"""

import base64
import json
import os
import sys
import urllib.request

REPO = "Hurley2017/OnDemandPivot"
BRANCH = "main"
FILE = "bundle/OnDemandPivot-Update.txt"

URLS = [
    "https://raw.githubusercontent.com/{r}/{b}/{f}",
    "https://github.com/{r}/raw/{b}/{f}",
    "https://api.github.com/repos/{r}/contents/{f}?ref={b}",
]


def download():
    """Return (url, bytes) for the first address that answers with a bundle."""
    problems = []
    for template in URLS:
        url = template.format(r=REPO, b=BRANCH, f=FILE)
        try:
            with urllib.request.urlopen(url, timeout=90) as response:
                data = response.read()
        except Exception as exc:                       # noqa: BLE001
            problems.append("  %-22s %s" % (type(exc).__name__, url))
            continue

        # The API answers with JSON carrying base64; the others with the file.
        if "api.github.com" in url:
            try:
                data = base64.b64decode(json.loads(data)["content"])
            except Exception as exc:                   # noqa: BLE001
                problems.append("  %-22s %s" % ("bad API payload", exc))
                continue

        if b"### FILE: " not in data or not data.startswith(b"#!/usr/bin/env"):
            problems.append("  %-22s %s" % ("not a bundle", url))
            continue

        return url, data

    raise SystemExit(
        "Could not fetch the update. Tried:\n" + "\n".join(problems) +
        "\n\nIf every address failed, the proxy is blocking command-line "
        "traffic to GitHub. The browser reaches github.com, so ask IT to allow "
        "raw.githubusercontent.com as well."
    )


def main():
    target = sys.argv[1] if len(sys.argv) > 1 else "OnDemandPivot"

    print("Fetching the latest update...")
    url, data = download()
    print("  %d bytes from %s" % (len(data), url))

    path = os.path.join(os.environ.get("TEMP", "."), "ondemandpivot-update.txt")
    with open(path, "wb") as handle:
        handle.write(data)
    print("  saved to %s" % path)

    print("Applying to %s\n" % os.path.abspath(target))
    # The bundle is a program; run it with the target as its argument.
    saved_argv = sys.argv
    sys.argv = [path, target]
    try:
        exec(compile(data, path, "exec"), {"__name__": "__main__"})
    finally:
        sys.argv = saved_argv
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
